# Individual stat actions retain their character-root behavior; editor routing handles All.
DI_ce_stats_init = {
    scope = character
    effect = {
        if = {
            limit = {
                OR = {
                    NOT = { has_variable = DI_ce_stats_character }
                    var:DI_ce_stats_character ?= { is_alive = no }
                }
            }
            set_variable = { name = DI_ce_stats_character value = this }
        }
    }
}
DI_ce_stats_select = {
    scope = character
    saved_scopes = { target }
    is_valid = { is_alive = yes }
    effect = {
        if = {
            limit = { is_alive = yes }
            save_scope_as = selected_character
            scope:target = {
                remove_variable = DI_ce_stats_all
                set_variable = { name = DI_ce_stats_character value = scope:selected_character }
            }
        }
    }
}
DI_ce_stats_is_immortal = {
    scope = character
    is_valid = { is_alive = yes is_immortal = yes }
}
DI_ce_stats_select_all = {
    scope = character
    effect = {
        set_variable = DI_ce_stats_all
    }
}
DI_ce_editor_dread_min = {
    scope = character
    is_valid = {
        OR = {
            has_variable = DI_ce_stats_all
            AND = {
                exists = var:DI_ce_stats_character
                var:DI_ce_stats_character = {
                    is_alive = yes
                    NOT = { dread = 0 }
                }
            }
        }
    }
    effect = {
        if = {
            limit = { has_variable = DI_ce_stats_all }
            every_in_list = {
                variable = pinned_characters
                add_dread = DI_ce_dread_min_delta
            }
        }
        else_if = {
            limit = { exists = var:DI_ce_stats_character }
            var:DI_ce_stats_character = {
                if = {
                    limit = {
                        is_alive = yes
                        NOT = { dread = 0 }
                    }
                    add_dread = DI_ce_dread_min_delta
                }
            }
        }
    }
}

DI_ce_stats_lock_dread = {
    scope = character
    effect = {
        if = {
            limit = { has_variable = DI_ce_stats_all }
            every_in_list = {
                variable = pinned_characters
                if = { limit = { NOT = { has_character_modifier = no_dread_change } } add_character_modifier = no_dread_change }
            }
        }
        else_if = {
            limit = { exists = var:DI_ce_stats_character }
            var:DI_ce_stats_character = { if = { limit = { NOT = { has_character_modifier = no_dread_change } } add_character_modifier = no_dread_change } }
        }
    }
}

            }

        else_if = {

            var:DI_ce_stats_character = {

                    add_stress = DI_ce_stress_min_delta

            }
        }
    }
