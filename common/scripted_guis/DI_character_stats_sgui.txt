# Stats act on the character passed as root, never implicitly on all pins.
DI_ce_stats_init = {
    scope = character
    effect = {
        if = {
            limit = {
                OR = {
                    NOT = { has_variable = DI_ce_stats_character }
                    var:DI_ce_stats_character ?= { is_alive = no }
                }
            }
            set_variable = { name = DI_ce_stats_character value = this }
        }
    }
}

DI_ce_stats_select = {
    scope = character
    saved_scopes = { target }
    is_valid = { is_alive = yes }
    effect = {
        if = {
            limit = { is_alive = yes }
            save_scope_as = selected_character
            scope:target = {
                set_variable = { name = DI_ce_stats_character value = scope:selected_character }
            }
        }
    }
}

DI_ce_stats_is_immortal = {
    scope = character
    is_valid = { is_alive = yes is_immortal = yes }
}

DI_ce_dread_min = {
    scope = character
    is_valid = {
        is_alive = yes
        NOT = { dread = 0 }
    }
    effect = {
        if = {
            limit = {
                is_alive = yes
                NOT = { dread = 0 }
            }
            add_dread = DI_ce_dread_min_delta
        }
    }
}
