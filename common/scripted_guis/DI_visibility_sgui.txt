### Multiplayer access control ###

# Called by the host client's GUI using the global IsHost function.
# Store a character scope so script-side access checks do not need lobby data.
DI_register_host_character_sgui = {
	scope = character

	is_valid = {
		OR = {
			NOT = { exists = global_var:DI_host_character }
			NOT = { this = global_var:DI_host_character }
		}
	}

	effect = {
		set_global_variable = {
			name = DI_host_character
			value = root
		}
	}
}

# Debug check: has the local player's character been registered as the host?
DI_host_character_check_sgui = {
	scope = character

	is_valid = {
		DI_is_host_character_trigger = yes
	}
}

# Valid when the character may use the menu / interactions under the current game rule.
DI_menu_access_sgui = {
	scope = character

	is_valid = {
		DI_can_access_menu_trigger = yes
	}
}

DI_interaction_access_sgui = {
	scope = character

	is_valid = {
		DI_can_access_interactions_trigger = yes
	}
}

# Closes the interaction cheat menu for characters that lost access (e.g. host change). Fired by the GUI state in DI_button.gui.
DI_revoke_interaction_access_sgui = {
	scope = character

	is_valid = {
		NOT = { DI_can_access_interactions_trigger = yes }
		any_owned_story = {
			story_type = DI_misc_variables_story
			has_variable = DI_cheat_menu
		}
	}

	effect = {
		random_owned_story = {
			limit = {
				story_type = DI_misc_variables_story
				has_variable = DI_cheat_menu
			}
			remove_variable = DI_cheat_menu
		}
	}
}
