# Rebuild from the loaded tradition database whenever the picker opens.
# Clear first so reopening or loading an existing save cannot duplicate entries.
DI_tradition_picker_init_sgui = {
    scope = character
    effect = {
        save_scope_as = DI_tradition_picker_owner
        DI_tradition_picker_clear_uses_effect = yes
        if = {
            limit = { has_global_variable_list = DI_tradition_picker_realm }
            clear_global_variable_list = DI_tradition_picker_realm
        }
        if = {
            limit = { has_global_variable_list = DI_tradition_picker_combat }
            clear_global_variable_list = DI_tradition_picker_combat
        }
        if = {
            limit = { has_global_variable_list = DI_tradition_picker_societal }
            clear_global_variable_list = DI_tradition_picker_societal
        }
        if = {
            limit = { has_global_variable_list = DI_tradition_picker_ritual }
            clear_global_variable_list = DI_tradition_picker_ritual
        }
        if = {
            limit = { has_global_variable_list = DI_tradition_picker_regional }
            clear_global_variable_list = DI_tradition_picker_regional
        }
        if = {
            limit = { has_global_variable_list = DI_tradition_picker_other }
            clear_global_variable_list = DI_tradition_picker_other
        }
        every_culture_tradition = {
            save_scope_as = tradition
            if = {
                limit = { has_tradition_category = realm }
                add_to_global_variable_list = { name = DI_tradition_picker_realm target = this }
                DI_tradition_picker_classify_use_effect = { CATEGORY = realm }
            }
            else_if = {
                limit = { has_tradition_category = combat }
                add_to_global_variable_list = { name = DI_tradition_picker_combat target = this }
                DI_tradition_picker_classify_use_effect = { CATEGORY = combat }
            }
            else_if = {
                limit = { has_tradition_category = societal }
                add_to_global_variable_list = { name = DI_tradition_picker_societal target = this }
                DI_tradition_picker_classify_use_effect = { CATEGORY = societal }
            }
            else_if = {
                limit = { has_tradition_category = ritual }
                add_to_global_variable_list = { name = DI_tradition_picker_ritual target = this }
                DI_tradition_picker_classify_use_effect = { CATEGORY = ritual }
            }
            else_if = {
                limit = { has_tradition_category = regional }
                add_to_global_variable_list = { name = DI_tradition_picker_regional target = this }
                DI_tradition_picker_classify_use_effect = { CATEGORY = regional }
            }
            else = {
                # Overflow only: none of the five vanilla categories matched.
                add_to_global_variable_list = { name = DI_tradition_picker_other target = this }
                DI_tradition_picker_classify_use_effect = { CATEGORY = other }
            }
        }
    }
}
