DI_tm_choose_county_filter = {
    scope = character
    saved_scopes = {
        filter_choice
    }
    effect = {
        save_scope_as = player

        remove_variable = DI_tm_filter_active_control
        remove_variable = DI_tm_filter_active_development
        remove_variable = DI_tm_filter_active_domain

        if = {
            limit = { scope:filter_choice = flag:DI_tm_filter_control }
            set_variable = DI_tm_filter_active_control
        }
        else_if = {
            limit = { scope:filter_choice = flag:DI_tm_filter_development }
            set_variable = DI_tm_filter_active_development
        }
        else_if = {
            limit = { scope:filter_choice = flag:DI_tm_filter_domain }
            set_variable = DI_tm_filter_active_domain
        }
        else = {
            add_gold = 100
        }

        DI_tm_refresh_counties_effect = yes
    }
}
DI_tm_move_realm_capital = {
    scope = character
    saved_scopes = { province barony }
    is_valid = { DI_tm_can_move_realm_capital = yes }
    effect = {
        if = {
            limit = { DI_tm_can_move_realm_capital = yes }
            scope:barony.holder = { set_realm_capital = scope:barony.county }
        }
    }
}

DI_tm_opinion_add = {
    scope = character
    is_valid = { any_in_list = { variable = DI_selected_titles_counties exists = holder } }
    effect = {
        DI_tm_refresh_counties_effect = yes
        every_in_list = { variable = DI_selected_titles_counties if = { limit = { NOT = { has_county_modifier = DI_tm_county_opinion } } add_county_modifier = { modifier = DI_tm_county_opinion } } }
    }
}

    }
}
