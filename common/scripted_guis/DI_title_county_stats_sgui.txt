DI_tm_choose_county_filter = {
    scope = character
    saved_scopes = {
        filter_choice
    }
    effect = {
        save_scope_as = player

        remove_variable = DI_tm_filter_active_control
        remove_variable = DI_tm_filter_active_prosperity
        remove_variable = DI_tm_filter_active_domain

        if = {
            limit = { scope:filter_choice = flag:DI_tm_filter_control }
            set_variable = DI_tm_filter_active_control
        }
        else_if = {
            limit = { scope:filter_choice = flag:DI_tm_filter_prosperity }
            set_variable = DI_tm_filter_active_prosperity
        }
        else_if = {
            limit = { scope:filter_choice = flag:DI_tm_filter_domain }
            set_variable = DI_tm_filter_active_domain
        }

        DI_tm_refresh_counties_effect = yes
    }
}

DI_tm_popular_opinion_apply = {
    scope = character
    saved_scopes = { operation step }
    is_valid = { any_in_list = { variable = DI_selected_titles_counties exists = holder } }
    effect = {
        DI_tm_refresh_counties_effect = yes
        every_in_list = {
            variable = DI_selected_titles_counties
            if = {
                limit = { exists = holder }
                if = {
                    limit = { scope:operation = flag:DI_tm_county_op_min }
                    add_popular_opinion = DI_tm_county_popular_opinion_min_delta
                }
                else_if = {
                    limit = { scope:operation = flag:DI_tm_county_op_subtract }
                    add_popular_opinion = DI_tm_county_popular_opinion_subtract_delta
                }
                else_if = {
                    limit = { scope:operation = flag:DI_tm_county_op_mid }
                    add_popular_opinion = DI_tm_county_popular_opinion_mid_delta
                }
                else_if = {
                    limit = { scope:operation = flag:DI_tm_county_op_add }
                    add_popular_opinion = DI_tm_county_popular_opinion_add_delta
                }
                else_if = {
                    limit = { scope:operation = flag:DI_tm_county_op_max }
                    add_popular_opinion = DI_tm_county_popular_opinion_max_delta
                }
            }
        }
        DI_tm_refresh_counties_effect = yes
    }
}
