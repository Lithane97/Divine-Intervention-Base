    OR = {
        DI_fe_tenets_require_pam_dlc_trigger = yes
        DI_fe_tenets_require_fp1_dlc_trigger = yes
        DI_fe_tenets_require_tgp_dlc_trigger = yes
    }
}

    OR = {
        AND = {
            DI_fe_tenets_require_pam_dlc_trigger = yes
            has_pam_dlc_trigger = yes
        }
        AND = {
            DI_fe_tenets_require_fp1_dlc_trigger = yes
            has_fp1_dlc_trigger = yes
        }
        AND = {
            DI_fe_tenets_require_tgp_dlc_trigger = yes
            has_tgp_dlc_trigger = yes
        }
        NOT = { DI_fe_tenet_requires_dlc_trigger = yes }
    }
