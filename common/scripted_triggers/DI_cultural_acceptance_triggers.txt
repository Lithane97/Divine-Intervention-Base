# Player scope. Always resolve the selected characters' current cultures.
DI_ca_has_characters = {
    exists = var:DI_innovation_selected_char
    exists = var:DI_innovation_selected_char_copy
    exists = var:DI_innovation_selected_char.culture
    exists = var:DI_innovation_selected_char_copy.culture
}
DI_ca_valid_pair = {
    DI_ca_has_characters = yes
    NOT = { var:DI_innovation_selected_char.culture = var:DI_innovation_selected_char_copy.culture }
}
DI_ca_can_min = {
    DI_ca_valid_pair = yes
    var:DI_innovation_selected_char.culture = { cultural_acceptance = { target = root.var:DI_innovation_selected_char_copy.culture value > 0 } }
}
DI_ca_can_subtract = {
    DI_ca_valid_pair = yes
    var:DI_innovation_selected_char.culture = { cultural_acceptance = { target = root.var:DI_innovation_selected_char_copy.culture value > 0 } }
}
DI_ca_can_mid = {
    DI_ca_valid_pair = yes
    var:DI_innovation_selected_char.culture = { NOT = { cultural_acceptance = { target = root.var:DI_innovation_selected_char_copy.culture value = 50 } } }
}
DI_ca_can_add = {
    DI_ca_valid_pair = yes
    var:DI_innovation_selected_char.culture = { cultural_acceptance = { target = root.var:DI_innovation_selected_char_copy.culture value < 100 } }
}
DI_ca_can_max = {
    DI_ca_valid_pair = yes
    var:DI_innovation_selected_char.culture = { cultural_acceptance = { target = root.var:DI_innovation_selected_char_copy.culture value < 100 } }
}
