### Character Editor title and authority helpers ###

DI_ce_apply_to_edit_targets_effect = {
    random_owned_story = {
        limit = {
            story_type = DI_misc_variables_story
        }
        save_scope_as = DI_story
    }

    if = {
        limit = {
            scope:DI_story = {
                has_variable = DI_pinned_char_var
            }
        }
        every_living_character = {
            limit = {
                has_character_flag = pinned_for_edit
            }
            $TARGET_EFFECT$ = yes
        }
    }
    else = {
        $TARGET_EFFECT$ = yes
    }
}

DI_ce_apply_to_pinned_characters_effect = {
    every_living_character = {
        limit = {
            has_character_flag = pinned_for_edit
        }
        $TARGET_EFFECT$ = yes
    }
}

DI_ce_take_realm_titles_target_effect = {
    save_scope_as = DI_ce_title_recipient

    top_liege ?= {
        every_sub_realm_title = {
            limit = {
                tier >= tier_county
                is_noble_family_title = no
                is_landless_type_title = no
                NOT = {
                    holder = scope:DI_ce_title_recipient
                }
            }
            scope:DI_ce_title_recipient = {
                DI_get_title_effect = { TITLE = prev }
            }
        }
    }
}

# Sets the scoped ruler's government authority law to the given LEVEL (0-3).
# Every authority law group with four levels is handled: Crown Authority (feudal, clan),
# Church Authority (theocracy, ecclesiastical), Imperial Bureaucracy (administrative),
# Tribal Authority, and the Celestial, Meritocratic and Japanese bureaucracies.
# Nomadic Authority has five levels and is not handled.
# Vassals whose law must match their liege's (should_synchronize_liege_law) are skipped.
DI_ce_set_crown_authority_target_effect = {
    if = {
        limit = {
            is_ruler = yes
            should_synchronize_liege_law = no
        }
        if = {
            limit = {
                OR = {
                    has_realm_law = crown_authority_0
                    has_realm_law = crown_authority_1
                    has_realm_law = crown_authority_2
                    has_realm_law = crown_authority_3
                }
                NOT = {
                    has_realm_law = crown_authority_$LEVEL$
                }
            }
            add_realm_law = crown_authority_$LEVEL$
        }
        else_if = {
            limit = {
                OR = {
                    has_realm_law = church_authority_0
                    has_realm_law = church_authority_1
                    has_realm_law = church_authority_2
                    has_realm_law = church_authority_3
                }
                NOT = {
                    has_realm_law = church_authority_$LEVEL$
                }
            }
            add_realm_law = church_authority_$LEVEL$
        }
        else_if = {
            limit = {
                OR = {
                    has_realm_law = imperial_bureaucracy_0
                    has_realm_law = imperial_bureaucracy_1
                    has_realm_law = imperial_bureaucracy_2
                    has_realm_law = imperial_bureaucracy_3
                }
                NOT = {
                    has_realm_law = imperial_bureaucracy_$LEVEL$
                }
            }
            add_realm_law = imperial_bureaucracy_$LEVEL$
        }
        else_if = {
            limit = {
                OR = {
                    has_realm_law = tribal_authority_0
                    has_realm_law = tribal_authority_1
                    has_realm_law = tribal_authority_2
                    has_realm_law = tribal_authority_3
                }
                NOT = {
                    has_realm_law = tribal_authority_$LEVEL$
                }
            }
            add_realm_law = tribal_authority_$LEVEL$
        }
        else_if = {
            limit = {
                OR = {
                    has_realm_law = celestial_bureaucracy_0
                    has_realm_law = celestial_bureaucracy_1
                    has_realm_law = celestial_bureaucracy_2
                    has_realm_law = celestial_bureaucracy_3
                }
                NOT = {
                    has_realm_law = celestial_bureaucracy_$LEVEL$
                }
            }
            add_realm_law = celestial_bureaucracy_$LEVEL$
        }
        else_if = {
            limit = {
                OR = {
                    has_realm_law = meritocratic_bureaucracy_0
                    has_realm_law = meritocratic_bureaucracy_1
                    has_realm_law = meritocratic_bureaucracy_2
                    has_realm_law = meritocratic_bureaucracy_3
                }
                NOT = {
                    has_realm_law = meritocratic_bureaucracy_$LEVEL$
                }
            }
            add_realm_law = meritocratic_bureaucracy_$LEVEL$
        }
        else_if = {
            limit = {
                OR = {
                    has_realm_law = japanese_bureaucracy_0
                    has_realm_law = japanese_bureaucracy_1
                    has_realm_law = japanese_bureaucracy_2
                    has_realm_law = japanese_bureaucracy_3
                }
                NOT = {
                    has_realm_law = japanese_bureaucracy_$LEVEL$
                }
            }
            add_realm_law = japanese_bureaucracy_$LEVEL$
        }
    }
}
