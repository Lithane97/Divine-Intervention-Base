# Character scope. Only player-controlled, unlanded non-rulers need adventurer
# preparation. AI characters and existing rulers keep their original behavior.
DI_prepare_title_recipient_effect = {
    if = {
        limit = {
            is_ai = no
            is_landed = no
            is_ruler = no
        }
        create_landless_adventurer_title_effect = {
            REASON = flag:voluntary
            FLAVOR_CHAR = this
        }
    }
}

# Capture TITLE before preparing the character: callers may pass prev, and the
# vanilla setup effect creates and scopes to its own adventurer title.
DI_get_title_effect = {
    $TITLE$ = { save_scope_as = DI_title_transfer_title }
    DI_prepare_title_recipient_effect = yes
    get_title = scope:DI_title_transfer_title
}
